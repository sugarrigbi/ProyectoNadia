struct Playlist {
    string Cancion;
    string Artista;
};

void MostrarPlaylist(const list<Playlist>& playlist) {
    cout << "\033[45m---------\033[0m" << endl;
    cout << "\033[45mPlaylist:\033[0m" << endl;
    cout << "\033[45m---------\033[0m" << endl;
    if (playlist.empty()) {
        cout << "\033[42m------------------------------------\033[0m" << endl;
        cout << "\033[42mLa lista de reproducción está vacía.\033[0m" << endl;
        cout << "\033[42m------------------------------------\033[0m" << endl;
        return;
    }

    for (const auto& Playlist : playlist) {
        cout << "\033[45m--\033[0m " << Playlist.Cancion << " -- " << Playlist.Artista << endl;
    }
    cout << endl;
}
