const int FILAS = 3;
const int COLUMNAS = 3;

//LLENAR DATOS
void llenarMatriz(int matriz[FILAS][COLUMNAS]) {
    cout << "\033[33mIngrese los valores para la matriz (" << FILAS << "x" << COLUMNAS << "):";
    for (int i = 0; i < FILAS; i++) {
      cout << "\n" << endl;
        for (int j = 0; j < COLUMNAS; j++) {
            cout << "\033[32mDato [" << i << "][" << j << "]:\033[0m ";
            cin >> matriz[i][j];
        }
    }
}

//MOSTRAR DATOS
void mostrarMatriz(int matriz[FILAS][COLUMNAS]) {
    cout << "\033[33m\nMatriz ingresada: " << endl;
    for (int i = 0; i < FILAS; i++) {
      cout << "|";
        for (int j = 0; j < COLUMNAS; j++) {
            cout << " " << matriz[i][j] << " " << "|";
        }
        cout << endl;
       cout << "------------" << "\n";
    } 
}

int main() {
    int matriz[FILAS][COLUMNAS];

  
    llenarMatriz(matriz);
    mostrarMatriz(matriz);

    return 0;
}
