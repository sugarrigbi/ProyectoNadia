void MostrarDatos(list<string> & sList){
    list<string>::iterator pos;
    pos = sList.begin();

    while (pos != sList.end())
        {
            cout << "- " << *pos << endl;
            pos++;
        }
}
void MostrarDatosReverse(list<string> & sList){
    list<string>::reverse_iterator pos;

    for (pos = sList.rbegin(); pos != sList.rend(); ++pos) {
        cout << "- " << *pos << endl;
    }
}
